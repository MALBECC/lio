# CMakeLists.txt for liosolo executables
enable_language(Fortran)

# Add the executable targets
add_executable(liosolo liosolo.f90)
add_executable(liomd liomd.f90)

# Fortran module output directory for these executables (optional, but good practice if they produced modules)
# If they only consume modules, this is less critical as include paths from lio-g2g handle finding .mod files.
# Let's assume for now they might produce modules or for consistency with lioamber.
set_target_properties(liosolo liomd PROPERTIES
    Fortran_MODULE_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}/obj
)
# Ensure this directory is part of their include path IF they produce modules consumed by each other
# or if you want a consistent place for .mod files.
# However, lio-g2g already propagates its module directory for consumption.
# For simple executables, this might be overkill unless they have internal modules.

# Link with required libraries
# g2g is transitively linked via lio-g2g (which links g2g PUBLIC)
target_link_libraries(liosolo PRIVATE lio-g2g ${EXTRA_LIBS})
target_link_libraries(liomd PRIVATE lio-g2g ${EXTRA_LIBS})

# Preprocessor definitions needed directly by liosolo.f90 / liomd.f90
# Linking is handled transitively by lio-g2g.
if(USE_CUDA AND (USE_CUBLAS OR USE_MAGMA))
    target_compile_definitions(liosolo PRIVATE CUBLAS) # If liosolo.f90 uses #ifdef CUBLAS
    target_compile_definitions(liomd PRIVATE CUBLAS)   # If liomd.f90 uses #ifdef CUBLAS
    if(USE_MAGMA)
        target_compile_definitions(liosolo PRIVATE magma) # If liosolo.f90 uses #ifdef magma
        target_compile_definitions(liomd PRIVATE magma)   # If liomd.f90 uses #ifdef magma
    endif()
endif()

# OpenMP support for liosolo.f90 / liomd.f90 themselves
if(USE_PARALLEL)
    find_package(OpenMP QUIET) # lio-g2g might have already found it
    if(OpenMP_Fortran_FOUND)
        target_link_libraries(liosolo PRIVATE OpenMP::OpenMP_Fortran)
        target_link_libraries(liomd PRIVATE OpenMP::OpenMP_Fortran)
    else()
        message(WARNING "OpenMP requested for liosolo/liomd but OpenMP_Fortran not found.")
    endif()
endif()

# Compiler-specific options FOR liosolo AND liomd
# These are for enabling preprocessing and specifying module output/search paths
# The search path for modules from lio-g2g is handled by lio-g2g's INTERFACE_INCLUDE_DIRECTORIES.
# These options here are more about how liosolo/liomd themselves are compiled.
set(LIO_SOLO_MODULE_DIR ${CMAKE_CURRENT_BINARY_DIR}/obj) # Define once
# Create the directory, as Fortran compilers might not do it.
file(MAKE_DIRECTORY ${LIO_SOLO_MODULE_DIR})

target_compile_options(liosolo PRIVATE
    # Intel Fortran compiler settings
    $<$<OR:$<Fortran_COMPILER_ID:IntelLLVM>,$<Fortran_COMPILER_ID:Intel>>:-fpp># -module ${LIO_SOLO_MODULE_DIR}>
    # GNU Fortran compiler settings
    $<$<Fortran_COMPILER_ID:GNU>:-cpp># -J${LIO_SOLO_MODULE_DIR}>
    $<$<AND:$<Fortran_COMPILER_ID:GNU>,$<VERSION_GREATER_EQUAL:${CMAKE_Fortran_COMPILER_VERSION},10.0>>:-fallow-argument-mismatch> # Use VERSION_GREATER_EQUAL
)
target_compile_options(liomd PRIVATE
    # Intel Fortran compiler settings
    $<$<OR:$<Fortran_COMPILER_ID:IntelLLVM>,$<Fortran_COMPILER_ID:Intel>>:-fpp># -module ${LIO_SOLO_MODULE_DIR}>
    # GNU Fortran compiler settings
    $<$<Fortran_COMPILER_ID:GNU>:-cpp># -J${LIO_SOLO_MODULE_DIR}>
    $<$<AND:$<Fortran_COMPILER_ID:GNU>,$<VERSION_GREATER_EQUAL:${CMAKE_Fortran_COMPILER_VERSION},10.0>>:-fallow-argument-mismatch> # Use VERSION_GREATER_EQUAL
)
# Note: The -J or -module path here tells the compiler where to PUT modules generated by
# liosolo.f90/liomd.f90 and also where to SEARCH for modules.
# For SEARCHING modules from lio-g2g, the include paths propagated by lio-g2g are key.
# This ${LIO_SOLO_MODULE_DIR} is primarily for modules *generated by* liosolo/liomd.
# If liosolo.f90 and liomd.f90 don't generate any .mod files,
# then -module / -J pointing to their own binary dir might be less critical,
# but -fpp / -cpp might still be needed if they use preprocessor directives.


# Installation
# ------------
include(GNUInstallDirs)
message(STATUS "Installing liosolo and liomd to ${CMAKE_INSTALL_BINDIR}")
install(TARGETS liosolo liomd
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
